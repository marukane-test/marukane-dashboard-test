/** * 丸兼林業 現場ダッシュボード（A案：Google Apps Script） * ----------------------------------------------------- * PlatioからエクスポートしたCSV（Google Driveの検証用フォルダに手動で置く）を読み込み、 * このスクリプトが紐づくGoogleスプレッドシートに集計結果を書き出す。 * このスプレッドシートをGoogleサイトに埋め込んで公開する想定（A案）。 * * 参照した仕様：2026年7月14日 NBC公式議事録（3社打ち合わせ）確定版 * * 使い方： * 1. 新規のGoogleスプレッドシートを開く * 2. 拡張機能 → Apps Script を開き、このファイルの内容を貼り付ける * 3. 下記 CONFIG のDRIVE_FOLDER_IDが検証用フォルダのIDになっているか確認 * 4. エディタ上部で関数 main を選び、実行（初回は権限の許可が必要） * 5. 元のスプレッドシートに戻り、タブが自動生成されているか確認 */ // ============================================================ // CONFIG：目標値・区分マッピングなど「変わりうる値」はすべてここに集約する // （値を変える場合は、このCONFIGブロックだけを直せばよい設計） // ============================================================ const CONFIG = { // Platio CSVを置いているGoogle Driveフォルダ（検証用） DRIVE_FOLDER_ID: '1LvRHqM9sKlpI92LB_LmsXm43xruOQGJr', // ダッシュボードで扱う現場（タブ）一覧。実績が0件の現場もタブとしては表示する。 // 2026年7月14日議事録に基づく確定リスト SITES: [ '設楽町段戸国有林', '豊田市刈萱町', '豊田市黒田町', '豊田市平岩町', 'その他', '豊田市大洞町', ], // ダッシュボードの4区分（進捗集計の対象） PROGRESS_CATEGORIES: ['伐倒', '集材', '造材', '運搬'], // Platioの「作業内容」の生の値 → ダッシュボード上の区分 のマッピング // 【暫定ルール・要確認事項】切り捨て伐倒は4区分の進捗集計から除外し、別枠で表示する。 // 材積は集計対象外だが、労働時間（人工）は生産性計算に含める。 // その他（林内/林外作業）・作業道作設・機械修理は「その他作業」として一括表示（進捗集計対象外）。 // → この区分けは酒井社長との合意待ち。次回ミーティングで要相談。 CATEGORY_MAP: { '伐倒': '伐倒', '集材': '集材', '造材': '造材', '運搬': '運搬', 'フォワーダー運搬': '運搬', '切り捨て伐倒': '切り捨て伐倒', 'その他（林内作業）': 'その他作業', 'その他（林外作業）': 'その他作業', '作業道作設': 'その他作業', '機械修理': 'その他作業', }, // 年間目標（搬出量ベース）。 // 【重要・未解決】伐倒・集材・造材・運搬は同じ木の工程のため単純合算はできない。 // 目標4,500㎥の進捗は「運搬（＝搬出）」区分の累計材積を代理指標として使う。 // この集計ルールの是非は酒井社長の判断待ち（8/10説明資料に明記すること）。 ANNUAL_TARGET_M3: 4500, ANNUAL_TARGET_CATEGORY: '運搬', // 搬出量の代理指標として運搬区分を使用 // 現場×月×区分ごとの計画値（㎥）。 // 【要入力】実数値は酒井社長・NBCとすり合わせ後に入力する。現状は検証用の仮値（0）。 // 構造：MONTHLY_TARGETS['現場名']['YYYY-MM']['区分名'] = 計画値(㎥) MONTHLY_TARGETS: { // 例（実データが揃ったら以下のように追記する）： // '豊田市刈萱町': { '2026-07': { '伐倒': 50, '集材': 20, '造材': 80, '運搬': 40 } }, }, // 1人工あたりの標準労働時間（h）。生産性（生産量÷人工）の換算に使用。 // 【要確認】この換算ルールは酒井社長に確認していない暫定値。 HOURS_PER_NINKU: 8, }; // ============================================================ // メイン処理 // ============================================================ function main() { updateSpreadsheetTitle_(); const csvFile = getLatestCsvFile_(CONFIG.DRIVE_FOLDER_ID); if (!csvFile) { throw new Error('指定フォルダにCSVファイルが見つかりません。DRIVE_FOLDER_IDを確認してください。'); } const rows = parseCsvFile_(csvFile); const entries = buildEntries_(rows); writeRawDataSheet_(entries); const summary = aggregate_(entries); writeSiteSheet_('全体', summary.overall); CONFIG.SITES.forEach(site => { writeSiteSheet_(site, summary.bySite[site] || emptySiteSummary_()); }); writeExecutiveSheet_(summary.overall); SpreadsheetApp.getActiveSpreadsheet().toast('ダッシュボードを更新しました：' + csvFile.getName()); } // ============================================================ // CSV読み込み // ============================================================ function getLatestCsvFile_(folderId) { const folder = DriveApp.getFolderById(folderId); const files = folder.getFilesByType(MimeType.CSV); let latest = null; while (files.hasNext()) { const f = files.next(); if (!latest || f.getLastUpdated() > latest.getLastUpdated()) { latest = f; } } return latest; } function parseCsvFile_(file) { let text = file.getBlob().getDataAsString('UTF-8'); // 先頭のBOM（文字化け対策で付与されることがある）を除去 if (text.charCodeAt(0) === 0xFEFF) { text = text.slice(1); } const data = Utilities.parseCsv(text); const header = data[0]; return data.slice(1).map(row => { const obj = {}; header.forEach((key, i) => { obj[key] = row[i]; }); return obj; }); } // ============================================================ // 行データ → エントリ展開（1行に作業が最大2件入っているため展開する） // ============================================================ function buildEntries_(rows) { const entries = []; rows.forEach(row => { const date = row['日付']; // 1つ目の作業 if (row['①作業内容']) { entries.push(makeEntry_( date, row['現場名'] || row['現場名（記入）'], row['①作業内容'], row['①材積（㎥)'] || row['①材積（㎥）'], row['①作業時間'] )); } // 2つ目の作業（ある場合のみ） // 【重要】実データで「2つ目の作業あり」がfalseなのに②作業内容が入っている行が見つかったため、 // フラグではなく②作業内容の有無で判定する（フラグを信用すると実績が漏れる） if (row['②作業内容']) { entries.push(makeEntry_( date, row['現場名2'] || row['現場名(記入)'], row['②作業内容'], row['②材積（㎥）'], row['②作業時間'] )); } }); return entries; } function makeEntry_(date, site, categoryRaw, volumeStr, hoursMsStr) { const category = CONFIG.CATEGORY_MAP[categoryRaw] || 'その他作業'; return { date: date, site: site || '(不明)', categoryRaw: categoryRaw, category: category, volume: parseFloat(volumeStr) || 0, hours: (parseFloat(hoursMsStr) || 0) / 3600000, // ミリ秒 → 時間 }; } // ============================================================ // 集計 // ============================================================ function emptySiteSummary_() { return { latestDate: null, weekStartDate: null, // 直近1週間の集計範囲（開始） weekEndDate: null, // 直近1週間の集計範囲（終了＝データ最終日） recentWeekByCategory: {}, // 【週次更新前提】直近1週間分の実績（旧：前日実績） cumulativeByCategory: {}, monthlyByCategory: {}, // { 'YYYY-MM': { category: {volume, hours} } } hoursByCategory: {}, // 生産性計算用（切り捨て伐倒の労働時間も含む） }; } // 'YYYY-MM-DD' の文字列からdays日前の日付文字列を返す function subtractDays_(dateStr, days) { const d = new Date(dateStr + 'T00:00:00'); d.setDate(d.getDate() - days); return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } function aggregate_(entries) { const bySite = {}; CONFIG.SITES.forEach(s => { bySite[s] = emptySiteSummary_(); }); // CSV内に仕様外の現場名が来た場合も拾えるように動的追加 const overall = emptySiteSummary_(); if (entries.length === 0) { return { overall, bySite }; } // 【週次更新前提】「前日実績」ではなく、データ最終日を含む直近7日間を1つの更新分として扱う const latestDate = entries.reduce((max, e) => (e.date > max ? e.date : max), entries[0].date); const weekStartDate = subtractDays_(latestDate, 6); // 最終日を含む7日間 overall.latestDate = latestDate; overall.weekStartDate = weekStartDate; overall.weekEndDate = latestDate; entries.forEach(e => { if (!bySite[e.site]) { bySite[e.site] = emptySiteSummary_(); // 未知の現場名にも対応 } [bySite[e.site], overall].forEach(target => { target.latestDate = latestDate; target.weekStartDate = weekStartDate; target.weekEndDate = latestDate; // 累計（4区分＋切り捨て伐倒＋その他作業） target.cumulativeByCategory[e.category] = (target.cumulativeByCategory[e.category] || 0) + e.volume; // 労働時間（生産性計算用。切り捨て伐倒も含めすべて加算） target.hoursByCategory[e.category] = (target.hoursByCategory[e.category] || 0) + e.hours; // 直近1週間（週次更新の1回分に相当）の実績 if (e.date >= weekStartDate && e.date <= latestDate) { target.recentWeekByCategory[e.category] = (target.recentWeekByCategory[e.category] || 0) + e.volume; } // 月次実績 const ym = e.date.slice(0, 7); // 'YYYY-MM' if (!target.monthlyByCategory[ym]) target.monthlyByCategory[ym] = {}; target.monthlyByCategory[ym][e.category] = (target.monthlyByCategory[ym][e.category] || 0) + e.volume; }); }); return { overall, bySite }; } // ============================================================ // シート出力：現場別タブ（「全体」タブも同じ形式で出力） // ============================================================ function writeSiteSheet_(name, summary) { const sheet = getOrCreateSheet_(name); sheet.clear(); sheet.getCharts().forEach(c => sheet.removeChart(c)); // 前回実行時のセル結合が残っていると書き込みエラーになるため解除しておく sheet.getRange(1, 1, sheet.getMaxRows(), sheet.getMaxColumns()).breakApart(); sheet.showColumns(1, sheet.getMaxColumns()); let row = 1; sheet.getRange(row, 1).setValue(name + '：現場ダッシュボード').setFontWeight('bold').setFontSize(14); row += 2; // --- 直近1週間の実績 / 累計実績（4区分）【週次更新前提】 --- const weekLabel = summary.weekStartDate && summary.weekEndDate ? summary.weekStartDate + '〜' + summary.weekEndDate : '-'; sheet.getRange(row, 1).setValue('直近1週間の実績（' + weekLabel + '）／累計実績').setFontWeight('bold'); row++; sheet.getRange(row, 1).setValue('データ最終更新日：' + (summary.latestDate || '-')).setFontStyle('italic'); row++; const header = ['区分', ...CONFIG.PROGRESS_CATEGORIES]; sheet.getRange(row, 1, 1, header.length).setValues([header]); row++; const weekRow = ['直近1週間実績', ...CONFIG.PROGRESS_CATEGORIES.map(c => summary.recentWeekByCategory[c] || 0)]; const cumRow = ['累計実績', ...CONFIG.PROGRESS_CATEGORIES.map(c => summary.cumulativeByCategory[c] || 0)]; sheet.getRange(row, 1, 1, header.length).setValues([weekRow]); row++; sheet.getRange(row, 1, 1, header.length).setValues([cumRow]); row += 2; // --- 参考：切り捨て伐倒／その他作業（進捗集計外・別枠表示） --- sheet.getRange(row, 1).setValue('参考（進捗集計には含まない）').setFontWeight('bold'); row++; sheet.getRange(row, 1, 1, 3).setValues([['区分', '累計材積(㎥)', '累計労働時間(h)']]); row++; ['切り捨て伐倒', 'その他作業'].forEach(cat => { sheet.getRange(row, 1, 1, 3).setValues([[ cat, summary.cumulativeByCategory[cat] || 0, summary.hoursByCategory[cat] || 0, ]]); row++; }); row++; // --- 月次：計画／実績（横長表：7/14議事録モックアップ準拠） --- // モックアップでは進捗率は月ごとではなく「合計」に対して区分ごとに1つだけ表示する sheet.getRange(row, 1).setValue('月次 計画／実績').setFontWeight('bold'); row++; const monthKeys = Object.keys(summary.monthlyByCategory).sort(); // ヘッダー1行目：区分名（計画・実績の2列を結合）、「月」は2行分結合 sheet.getRange(row, 1, 2, 1).merge().setValue('月').setVerticalAlignment('middle'); let headerCol = 2; CONFIG.PROGRESS_CATEGORIES.forEach(cat => { sheet.getRange(row, headerCol, 1, 2).merge().setValue(cat).setHorizontalAlignment('center').setFontWeight('bold'); headerCol += 2; }); row++; // ヘッダー2行目：計画／実績 headerCol = 2; CONFIG.PROGRESS_CATEGORIES.forEach(() => { sheet.getRange(row, headerCol).setValue('計画').setHorizontalAlignment('center').setFontWeight('bold'); sheet.getRange(row, headerCol + 1).setValue('実績').setHorizontalAlignment('center').setFontWeight('bold'); headerCol += 2; }); row++; const totals = {}; CONFIG.PROGRESS_CATEGORIES.forEach(c => { totals[c] = { target: 0, actual: 0 }; }); monthKeys.forEach(ym => { const values = [ym]; CONFIG.PROGRESS_CATEGORIES.forEach(cat => { const actual = summary.monthlyByCategory[ym][cat] || 0; const target = getMonthlyTarget_(name, ym, cat); totals[cat].target += target; totals[cat].actual += actual; values.push(target, actual); }); sheet.getRange(row, 1, 1, values.length).setValues([values]); row++; }); // 合計行 const totalValues = ['合計']; CONFIG.PROGRESS_CATEGORIES.forEach(cat => { totalValues.push(totals[cat].target, totals[cat].actual); }); sheet.getRange(row, 1, 1, totalValues.length).setValues([totalValues]).setFontWeight('bold'); row++; // 進捗率行（区分ごとに1つだけ、計画・実績の2列をまたいで表示） sheet.getRange(row, 1).setValue('進捗率'); let pctCol = 2; CONFIG.PROGRESS_CATEGORIES.forEach(cat => { const pct = totals[cat].target > 0 ? Math.round((totals[cat].actual / totals[cat].target) * 1000) / 10 : null; sheet.getRange(row, pctCol, 1, 2).merge().setValue(pct === null ? '-' : pct + '%').setHorizontalAlignment('center'); pctCol += 2; }); row += 2; // --- 進捗率バー（区分ごとに個別のバーチャート：モックアップ準拠の4区分別ボックス表示） --- sheet.getRange(row, 1).setValue('進捗率（累計・区分別）').setFontWeight('bold'); row++; const chartStartRow = row; const chartDataCol = 12; // 表示の邪魔にならない列（L列以降）に一時データを置き、あとで隠す CONFIG.PROGRESS_CATEGORIES.forEach((cat, i) => { const pct = totals[cat].target > 0 ? Math.round((totals[cat].actual / totals[cat].target) * 1000) / 10 : 0; sheet.getRange(chartStartRow + i, chartDataCol).setValue(cat); sheet.getRange(chartStartRow + i, chartDataCol + 1).setValue(pct); }); CONFIG.PROGRESS_CATEGORIES.forEach((cat, i) => { const dataRange = sheet.getRange(chartStartRow + i, chartDataCol, 1, 2); const chart = sheet.newChart() .setChartType(Charts.ChartType.BAR) .addRange(dataRange) .setPosition(row, 1, 0, 0) .setOption('title', cat) .setOption('legend', { position: 'none' }) .setOption('hAxis', { minValue: 0, maxValue: Math.max(100, totals[cat].target > 0 ? Math.round((totals[cat].actual / totals[cat].target) * 100) : 100) }) .setOption('width', 480) .setOption('height', 70) .build(); sheet.insertChart(chart); row += 3; // グラフ1つ分の縦スペース }); sheet.hideColumns(chartDataCol, 2); row += 2; // --- 生産量／人工／生産性 --- sheet.getRange(row, 1).setValue('生産量／人工／生産性（累計・全区分合算）').setFontWeight('bold'); row++; const totalVolume = Object.values(summary.cumulativeByCategory).reduce((a, b) => a + b, 0); const totalHours = Object.values(summary.hoursByCategory).reduce((a, b) => a + b, 0); const ninku = totalHours / CONFIG.HOURS_PER_NINKU; const productivity = ninku > 0 ? Math.round((totalVolume / ninku) * 100) / 100 : 0; sheet.getRange(row, 1, 1, 3).setValues([['生産量(㎥)', '人工', '生産性(㎥/人工)']]); row++; sheet.getRange(row, 1, 1, 3).setValues([[totalVolume, Math.round(ninku * 10) / 10, productivity]]); } function getMonthlyTarget_(site, ym, category) { const siteTargets = CONFIG.MONTHLY_TARGETS[site]; if (!siteTargets || !siteTargets[ym]) return 0; return siteTargets[ym][category] || 0; } // ============================================================ // シート出力：経営者ページ（年間目標4,500㎥の進捗） // ============================================================ function writeExecutiveSheet_(overallSummary) { const sheet = getOrCreateSheet_('経営者ページ'); sheet.clear(); sheet.getCharts().forEach(c => sheet.removeChart(c)); let row = 1; sheet.getRange(row, 1).setValue('経営者ページ：年間目標の進捗').setFontWeight('bold').setFontSize(14); row += 2; sheet.getRange(row, 1).setValue( '【留意点】伐倒・集材・造材・運搬は同じ木の工程のため単純合算できません。' + 'この進捗は「運搬（＝搬出量）」区分の累計を代理指標として使っています。' + '集計ルールの是非は酒井社長の確認待ちです。' ).setFontStyle('italic'); row += 2; const actual = overallSummary.cumulativeByCategory[CONFIG.ANNUAL_TARGET_CATEGORY] || 0; const target = CONFIG.ANNUAL_TARGET_M3; const pct = target > 0 ? Math.round((actual / target) * 1000) / 10 : 0; sheet.getRange(row, 1, 1, 3).setValues([['年間目標(㎥)', '累計実績(㎥・搬出量ベース)', '進捗率(%)']]); row++; sheet.getRange(row, 1, 1, 3).setValues([[target, actual, pct]]); } // ============================================================ // シート出力：生データ（検証用。パース結果の目視確認に使う） // ============================================================ function writeRawDataSheet_(entries) { const sheet = getOrCreateSheet_('生データ（検証用）'); sheet.clear(); sheet.getRange(1, 1, 1, 6).setValues([['日付', '現場名', '区分（変換後）', '区分（元の値）', '材積(㎥)', '労働時間(h)']]); if (entries.length === 0) return; const values = entries.map(e => [e.date, e.site, e.category, e.categoryRaw, e.volume, e.hours]); sheet.getRange(2, 1, values.length, 6).setValues(values); } // ============================================================ // ユーティリティ // ============================================================ function getOrCreateSheet_(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); } return sheet; } function updateSpreadsheetTitle_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm'); ss.rename('丸兼林業ダッシュボード（検証用）＿最終更新 ' + timestamp); }
